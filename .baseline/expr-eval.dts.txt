//// adapter.d.ts
import { ExpressionAdapter } from '@vueform/core';
/**
 * expr-eval expression adapter: builds the low-level `ExpressionEngine`
 * (single-expression evaluation and path extraction) — the `{…}` chain format,
 * warnings, and normalization are owned by core's `createExpression`.
 * Registered automatically by the `@vueform/expr-eval` plugin, or manually via
 * `config.$expression.adapter = exprEvalAdapter`.
 */
export declare const exprEvalAdapter: ExpressionAdapter;
/**
 * @deprecated Use `exprEvalAdapter`.
 */
export declare const $expression: ExpressionAdapter;

//// functions.d.ts
import { FormContext, VueformInstance } from '@vueform/core';
/**
 * The context expression functions read from: the owning Vueform instance (set
 * once by the adapter, used to resolve module doorways like date and locale
 * against the right Vueform instance) plus per-evaluation state bound before
 * each `evaluate()` call.
 */
export type ExpressionFunctionContext = {
    vueform?: VueformInstance;
    form$?: FormContext;
    el$?: any;
};
