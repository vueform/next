import type { UserConfig } from 'vite';
/**
 * Options for the `adapter` config factory.
 */
export interface AdapterOptions {
    /**
     * The package's published name (e.g. `'@vueform/ui'`) — used as the lib name,
     * added to externals, and source-relative self references in emitted `.d.ts`
     * are rewritten to it.
     */
    name: string;
    /**
     * The package directory. Pass `fileURLToPath(new URL('.', import.meta.url))`
     * from the package's `vite.config.ts`: tools that load workspace configs from
     * elsewhere (e.g. `vp run` building its task graph) do not set
     * `process.cwd()` to the package dir, which remains the fallback.
     */
    dir?: string;
    /**
     * Alias pointing at the source directory (e.g. `'@'`). No alias is registered
     * when omitted.
     */
    alias?: string;
    /**
     * Additional external module ids. `'vue'`, `name` and `@vueform/core` (all
     * subpaths) are always external.
     */
    external?: (string | RegExp)[];
    /**
     * Also emit minified IIFE bundles for `<script>`/CDN use, one per theme
     * (every `src/themes/<name>.ts` with a matching `src/<name>.ts` entry):
     * `dist/<name>.global.js` with its stylesheet beside it as `dist/<name>.css`,
     * everything bundled except `vue` (global `Vue`).
     */
    cdn?: boolean;
    /**
     * Also emit the vapor-mode build after the main bundle: the `src/components`
     * tree into `dist/vapor/components` plus a `dist/vapor/<theme>.mjs` entry per
     * theme, with every SFC compiled by Vue 3.6's vapor compiler. Shared logic is
     * not duplicated — vapor modules import it from the main dist via relative
     * paths, and `@vueform/core` component subpaths are rewritten to their
     * `/vapor` equivalents so the two mode trees never mix. Types are not
     * re-emitted; vapor exports reuse the `.d.ts` files from the main dist.
     */
    vapor?: boolean;
    /**
     * Additional Vite config merged (via `mergeConfig`) on top of the generated
     * one — an escape hatch for anything not covered by the options.
     */
    vite?: UserConfig;
}
/**
 * Vite config factory for adapter packages (ui, primevue, …) — usable directly
 * as the default export of a `vite.config.ts`:
 *
 * ```ts
 * import { createAdapterConfig } from '@vueform/utils'
 *
 * export default createAdapterConfig({ name: '@vueform/ui' })
 * ```
 *
 * The build mirrors `src` 1:1 into `dist` (what is public API is decided solely
 * by the package's `exports` map), ships the theme scss forwarders compiled
 * into css chunks plus the flattened `theme.css`, and keeps everything an
 * adapter builds on — vue, core, its own runtime dependencies — external.
 */
export declare function createAdapterConfig(options: AdapterOptions): Promise<UserConfig>;

/**
 * Resolves the `#shorthands` marker (a type-only import in component
 * `unstyled.ts` files) to an empty, side-effect-free virtual module, so Rollup
 * tree-shakes the import away entirely. The type side is covered by the ambient
 * `declare module '#shorthands'` in the root tsconfig's include.
 */
export declare function shorthandsMarker(): Plugin;
/**
 * Drops two kinds of dead chunks that `preserveModules` keeps around:
 *
 * - Empty chunks: type-only source modules (interfaces, type aliases) compile to
 *   empty `.mjs` files — their `.d.ts` is the real output. Nothing imports
 *   them, since Rollup already tree-shakes imports of empty side-effect-free
 *   modules.
 * - Orphaned SFC facades (`VfButton2.vue.mjs`): plugin-vue splits an SFC into a
 *   facade module and a script sub-module. Rollup traces re-exports to the
 *   script sub-module (which claims the `.vue.mjs` chunk name), leaving the
 *   facade as a pure default re-export chunk that nothing imports.
 */
/**
 * Restores consumer-side namespace reification under Rolldown (vite 8).
 *
 * `import * as ns from './barrel'` across preserved modules makes Rolldown
 * materialize the namespace as an `__exportAll({...})` object INSIDE the barrel
 * chunk and export it (`components_exports`). That extra export eagerly
 * references every binding of the barrel, so consumers bundling any import of
 * the barrel lose all tree-shaking (measured: the minimal ui bundle grew ~5x).
 * Rollup instead reified the namespace in the consuming chunk, keeping the
 * barrel clean.
 *
 * This pass rewrites the emitted chunks back to the Rollup shape:
 *
 * - consuming chunks: `import { x_exports } from "spec"` becomes `import * as
 *   x_exports from "spec"` (named siblings are kept),
 * - defining chunks: the `var x_exports = __exportAll({...});` block, the
 *   `x_exports` entry in the final export statement, and the then-unused
 *   `__exportAll` import are removed.
 */
export declare function namespaceExports(): Plugin;
export declare function pruneModules(): Plugin;
/**
 * Externalizes relative imports of the package's own theme chunk sources
 * (`src/scss/<chunk>.scss`) to the published `<name>/<chunk>.css` subpaths, so
 * source can reference the real scss files while dist modules carry the bare
 * self-referencing specifiers the exports map serves (real CSS for bundlers,
 * Node shims otherwise). The chunks themselves are compiled separately
 * (`themeScss`) — they never enter the bundle graph.
 */
export declare function scssExternals(name: string, scssAbs: string): Plugin;
/**
 * Copies the theme scss sources into dist verbatim and compiles each 1:1 to a
 * css chunk, with a Node exports-condition shim beside every file. Mixin-only
 * partials that compile to empty css keep their scss copy but get no css chunk
 * — their rules ship inside the chunks that `@use` them. `theme.scss` is the
 * aggregate recipe — handled by `themeCss`, never compiled or copied as a
 * chunk. `importMap` resolves package-specifier `@use`/`@forward` in the
 * sources (sass has no node resolution of its own).

export declare function themeScss(root: string, srcScss: string, outScss: string, outCss: string, importMap?: Record<string, string>): Plugin;
/**

export declare function themeCss(root: string, recipe: string, chunksDir: string, out: string, recipeOut: string): Plugin;

//// size/types.d.ts
/**
 * What a package's size suite measures, derived from its manifest:
 *
 * - `core` (`@vueform/core`): the building blocks — per-module, per-rule and
 *   per-locale costs. Adapter-independent (adapters re-export them unchanged),
 *   so they are measured once here and embedded into every adapter report.
 * - `adapter` (`vueform.group: "adapters"`): the bare core entry (`createVueform`
 *   imported from the adapter), per-component and per-theme costs.
 * - `plugin` (`vueform.group: "plugins"`): what the plugin adds on top of a full
 *   `@vueform/ui` build, split into Vueform code and its third-party
 *   dependency.
 */
export type MeasurementType = 'core' | 'adapter' | 'plugin';
export type ScenarioGroup = 'fixed'
/**
 * Reference build used only as the base of delta calculations (e.g. the
 * per-component reference setups), not shown in the report's intro table.
 */
 | 'ref' | 'component-remove' | 'theme' | 'module' | 'rule' | 'locale' | 'plugin' | 'pv';
export interface Scenario {
    name: string;
    group: ScenarioGroup;
    /**
     * Component the scenario measures (component-remove group).
     */
    component?: string;
    /**
     * Building block the scenario measures (module / rule / locale groups).
     */
    item?: string;
    /**
     * Name of the scenario this scenario's delta is measured against. Defaults to
     * `core` for building-block scenarios; the `model` module overrides it
     * because the core base already contains model.
     */
    base?: string;
    /**
     * Plugin the scenario measures (plugin group).
     */
    plugin?: string;
    /**
     * The plugin's heavy third-party dependency (plugin group), used in report
     * prose and headers.
     */
    dep?: string;
    /**
     * Name of the matched reference scenario containing only third-party code
     * (raw primevue/* modules, a plugin's dependency), used to split a scenario's
     * delta into a third-party part and a Vueform part.
     */
    pv?: string;
    /**
     * Human description used in reports.
     */
    description: string;
    /**
     * Whether the scenario registers the theme module. Reported sizes count CSS
     * assets only when it does — without the theme module the CSS is not
     * functional, so it is not part of the scenario's meaningful cost.
     */
    themed?: boolean;
    /**
     * Source code of the generated fixture src/main.ts.
     */
    entry: string;
}

import type { UserConfig } from 'vite';
/**
 * Options for the `vitest` config factory.
 */
export interface VitestOptions {
    /**
     * The package's own source alias (e.g. `'@'`), pointing at its `src`
     * directory. No alias is registered when omitted.
     */
    alias?: string;
    /**
     * Additional Vitest config merged (via `mergeConfig`) on top of the generated
     * one — an escape hatch for anything not covered by the defaults.
     */
    vite?: VitestConfig;
}
/**
 * Vitest's config shape: vite's `UserConfig` plus the `test` block, typed
 * loosely so this package needs no vitest dependency.
 */
export type VitestConfig = UserConfig & {
    test?: Record<string, any>;
};
/**
 * Vitest config factory for the workspace packages — usable directly as the
 * default export of a `vitest.config.ts`:
 *
 * ```ts
 * import { createVitestConfig } from '@vueform/utils'
 *
 * export default createVitestConfig()
 * ```
 *
 * Provides the shared testing setup with no per-package config: the Vue plugin,
 * happy-dom with vitest globals, the `integration` test project, and source
 * aliases for every workspace package (short `#<dir>` aliases with
 * exports-derived subpaths, plus full package names) so tests always run
 * against sources, never dist.
 *
 * A separate `bench` project picks up `tests/bench/**\/*.bench.ts` for `vitest
 * bench` — plain `vitest` never runs bench files and `vitest bench` never runs
 * tests, each mode matching exactly one project.
 *
 * When the package has a `tests/dts` directory, a `test-dts` project is added
 * that typechecks `tests/dts/**\/*.test-d.ts` with `vue-tsc` (run it alone with
 * `vitest run --project test-dts`). The directory needs its own `tsconfig.json`
 * covering the test files (the checker resolves imports through tsconfig paths,
 * not vite aliases), because the package tsconfig excludes `tests/dts` — type
 * tests are the `test-dts` gate's concern, not `check`'s.
 *
 * A `tests/dts-dist` directory likewise adds a `test-dts-dist` project, whose
 * tsconfig empties the root `paths` so package imports resolve through the
 * `exports` maps to built dist types — the published consumer's view. It needs
 * a prior build and is the fast d.ts gate for toolchain changes.
 */
export declare function createVitestConfig(options?: VitestOptions): Promise<VitestConfig>;
