//// module.d.mts
import * as _nuxt_schema from '@nuxt/schema';
import * as _vueform_primevue_components_names from '@vueform/primevue/components/names';
import * as _vueform_ui_components_names from '@vueform/ui/components/names';
import { BuiltInRules, UserConfig } from '@vueform/core';

type UiComponentName = _vueform_ui_components_names.ComponentNames;
type PrimeVueComponentName = _vueform_primevue_components_names.ComponentNames;
type VueformRuleName = BuiltInRules;
interface RuleFilter {
    include?: VueformRuleName[];
    exclude?: VueformRuleName[];
}
type InstallAdapterMode = 'prompt' | 'auto' | 'never';
/**
 * Options shared by every adapter.
 */
interface SharedModuleOptions<TComponentName extends string = string> {
    /**
     * Path of the user's Vueform config file relative to the project root,
     * without extension. Defaults to `./vueform.config` (`.ts` or `.js`).
     */
    configPath?: string;
    /**
     * Whether the adapter's stylesheets are included. Defaults to `true`.
     */
    css?: boolean;
    /**
     * Locales to register. The first one becomes the initial locale. Defaults to
     * `['en_US']`.
     */
    locales?: string[];
    /**
     * Components to register. Defaults to every component of the adapter.
     */
    include?: TComponentName[];
    /**
     * Components to leave out of the default set. Ignored when `include` is
     * given.
     */
    exclude?: TComponentName[];
    /**
     * Validation rules to register — a list, or include/exclude filters over the
     * adapter's full set. Defaults to every rule.
     */
    rules?: VueformRuleName[] | RuleFilter;
    /**
     * Inline Vueform config, merged under the config file's values.
     */
    config?: UserConfig;
    /**
     * What to do when the adapter package is not installed: ask before installing
     * it (`prompt`, the default), install it without asking (`auto`), or fail
     * with the install command (`never`). Prompting and auto-installing only
     * happen in an interactive dev session; otherwise the module fails with the
     * install command.
     */
    installAdapter?: InstallAdapterMode;
}
interface UiModuleOptions extends SharedModuleOptions<UiComponentName> {
    adapter: 'ui';
    /**
     * A built-in theme name (`origin`, `material`) — or a module specifier whose
     * default export is the token map (`~/themes/midnight`, a package name, or a
     * path relative to the project root).
     */
    theme?: 'origin' | 'material' | (string & {});
}
interface PrimeVueModuleOptions extends SharedModuleOptions<PrimeVueComponentName> {
    adapter: 'primevue';
}
type ModuleOptions = UiModuleOptions | PrimeVueModuleOptions;
type AdapterName = ModuleOptions['adapter'];

declare const _default: _nuxt_schema.NuxtModule<ModuleOptions, ModuleOptions, false>;

export { _default as default };
export type { AdapterName, InstallAdapterMode, ModuleOptions, PrimeVueModuleOptions, RuleFilter, SharedModuleOptions, UiModuleOptions, VueformRuleName };
