//// args.d.ts
/**
 * Minimal argument parser: splits `args` into positionals and `--flag`s.
 * `--flag=value` always captures the value; a flag listed in
 * `options.valueFlags` also consumes the next argument as its value (`--flag
 * value`); any other `--flag` is `true`.
 */
export interface ParsedArgs {
    positionals: string[];
    flags: Record<string, string | boolean>;
}
export declare function parseArgs(args: string[], options?: {
    valueFlags?: string[];
}): ParsedArgs;

//// autoregister.d.ts
export type RegisterPlan = {
    status: 'impossible';
    reason: string;
} | {
    status: 'registered';
    rel: string;
} | {
    status: 'possible';
    rel: string;
    preview: string[];
    apply(): void;
};
export type UnregisterPlan = {
    status: 'not-registered';
} | {
    status: 'impossible';
    reason: string;
} | {
    status: 'possible';
    rel: string;
    preview: string[];
    apply(): void;
};
/**
 * The manual-registration snippet for the barrel at `dir` (cwd-relative). The
 * import specifier is written relative to the project's entry file — where the
 * snippet gets pasted — falling back to the cwd-relative form when no entry is
 * found.
 */
export declare function registrationSnippet(cwd: string, dir: string): string;
/**
 * Decides whether the project's entry file can be safely edited to register the
 * components barrel, without applying anything. Returns one of:
 *
 * - `{ status: 'registered' }` — the barrel is already imported there
 * - `{ status: 'impossible', reason }` — a safety gate failed; don't ask
 * - `{ status: 'possible', rel, preview, apply }` — `apply()` performs the two
 *   minimal text insertions (import + spread), preserving formatting
 *
 * Gates: exactly one `createVueform(...)` call, argument absent or an inline
 * object literal, `components` (when present) a plain object literal, and the
 * `localComponents` identifier unused.
 */
export declare function planAutoRegister(cwd: string, componentsDir: string): Promise<RegisterPlan>;
/**
 * Decides whether the entry file's automatic barrel registration can be safely
 * removed. Returns one of:
 *
 * - `{ status: 'not-registered' }` — no barrel import in the entry file
 * - `{ status: 'impossible', reason }` — found, but not safely removable
 * - `{ status: 'possible', rel, preview, apply }` — `apply()` deletes the import
 *   and the `...<ident>` spread (or the whole `components` property when the
 *   spread is its only member)
 */
export declare function planUnregister(cwd: string, componentsDir: string): Promise<UnregisterPlan>;

//// barrel.d.ts
/**
 * (Re)generates the CLI-owned barrel in the components directory: it imports
 * every copied Vf*.vue and exports them as one `components` map — the single
 * thing a project registers. Always rewritten in full (never hash-tracked or
 * conflict-checked), so copy runs keep it in sync with the directory. Returns
 * the barrel's filename, or null when the directory has no copies.
 */
export declare function writeBarrel(dirAbs: string, useTs: boolean): "index.ts" | "index.js" | null;

//// config.d.ts
import type { CloneEntry, CopyEntry } from './types.js';
export declare const CONFIG_FILENAME = "vueform.config.json";
/**
 * Reads `componentsDir` out of a vueform.config.json. Returns null when the
 * file is missing, unreadable, or doesn't hold the key.
 */
export declare function readComponentsDir(configPath: string): string | null;
export declare function componentsDirSnippet(dir: string): string;
/**
 * Stores `componentsDir` in the config file, creating it when missing and
 * merging into it otherwise. Returns 'created', 'updated', or 'manual' when the
 * existing file isn't valid JSON.
 */
export declare function saveComponentsDir(configPath: string, dir: string): "manual" | "updated" | "created";
/**
 * Records copied components' provenance under the config's `copies` key (source
 * package, version, per-file hashes), merging with existing entries. `entries`
 * is a map of component name → entry. Returns 'manual' when the config file
 * isn't valid JSON.
 */
export declare function recordCopies(configPath: string, entries: Record<string, CopyEntry>): "manual" | "updated" | "created";
/**
 * Records a cloned element under the config's `clones` key (base component,
 * type, source package/version, per-file hashes). Returns 'manual' when the
 * config file isn't valid JSON.
 */
export declare function recordClone(configPath: string, name: string, entry: CloneEntry): "manual" | "updated" | "created";
/**
 * Removes entries from one of the config's provenance maps (`copies` or
 * `clones`). Drops the key entirely when empty. Returns true when the file was
 * rewritten; false when nothing was tracked (or the config is missing or
 * invalid).
 */
export declare function forgetEntries(configPath: string, key: 'copies' | 'clones', names: string[]): boolean;
/**
 * Reads one top-level config value; null when the file is missing/invalid.
 */
export declare function getConfigValue(configPath: string, key: string): {} | null;
/**
 * Sets one top-level config value, merging with the existing file. Returns
 * 'manual' when the file isn't valid JSON.
 */
export declare function setConfigValue(configPath: string, key: string, value: unknown): "manual" | "updated" | "created";
/**
 * Recommends a components directory: everything the CLI generates lives under a
 * single `vueform` directory in the project (components today, other generators
 * later) — `src/vueform/components` for Vite / src-based projects,
 * `vueform/components` otherwise (Nuxt, no src).
 */
export declare function recommendDir(cwd: string): "vueform/components" | "src/vueform/components";

//// settle.d.ts
/**
 * Formats the given files (absolute paths) with the project's prettier — only
 * when prettier is installed AND an explicit project config resolves for the
 * file (factory defaults are never applied: without a config there is nothing
 * safe to match). Only the passed files are touched, never anything else in the
 * project. Returns the number of files rewritten; failures are silent — the
 * copies are already correct, settling is cosmetic.
 */
export declare function settleFiles(cwd: string, absPaths: string[]): Promise<number>;
/**
 * Runs the user's `postCopy` command from vueform.config.json once per copy
 * run, from the project root. `{files}` in the command is replaced with the
 * shell-quoted list of written files (project-relative); without the
 * placeholder the command runs as-is. Returns true when the command exited 0.
 */
export declare function runPostCopy(cwd: string, command: string, relPaths: string[]): boolean;

//// types.d.ts
import type { ParsedArgs } from './args.js';
import type { Output } from './output.js';
/**
 * Canonical registry shapes — mirrors the JSON Schemas in
 * `packages/core/schemas/` (registry.json for the index, registry-item.json for
 * manifests). Update together with a `schemaVersion` bump.
 */
export interface RegistryIndexEntry {
    name: string;
    type: 'component';
    description?: string;
    element?: boolean;
}
export interface RegistryIndex {
    $schema?: string;
    schemaVersion: number;
    package: string;
    version: string;
    items: RegistryIndexEntry[];
}
export interface RegistryFile {
    path: string;
    target: string;
    kind: 'component' | 'types' | 'style';
    optional?: boolean;
    hash: string;
}
export interface RegistryClone {
    path: string;
    blueprint: string;
    hash: string;
    docs?: string;
}
export interface RegistryItem {
    $schema?: string;
    schemaVersion: number;
    name: string;
    type: 'component';
    package: string;
    version: string;
    description?: string;
    files: RegistryFile[];
    imports?: string[];
    dependencies?: Record<string, string>;
    registryDependencies?: string[];
    styles?: string[];
    register?: {
        component: string;
        import: string;
    };
    docs?: string;
    element?: boolean;
    clone?: RegistryClone;
}
/**
 * A successfully located adapter registry.
 */
export interface Registry {
    /**
     * Discriminant vs RegistryError — commands branch on `registry.error`.
     */
    error?: undefined;
    package: string;
    dir: string;
    index: RegistryIndex;
}
/**
 * `loadRegistry` failure surface.
 */
export interface RegistryError {
    error: 'no-package' | 'no-registry' | 'invalid' | 'version';
    package?: string;
    found?: unknown;
}
/**
 * vueform.config.json — the CLI's own config file (mirrors schema.json).
 */
export interface CopyEntry {
    package: string;
    version: string;
    files: Record<string, {
        hash: string;
        registryHash?: string;
    }>;
}
export interface CloneEntry {
    base: string;
    type: string;
    package: string;
    version: string;
    files: Record<string, {
        hash: string;
    }>;
}
export interface CliConfig {
    $schema?: string;
    componentsDir?: string;
    autoRegister?: boolean;
    postCopy?: string;
    copies?: Record<string, CopyEntry>;
    clones?: Record<string, CloneEntry>;
    [key: string]: unknown;
}
/**
 * Context handed to every command's `run`.
 */
export interface CommandContext extends ParsedArgs {
    out: Output;
}
/**
 * One CLI command's definition (usage line, flags, runner).
 */
export interface Command {
    name: string;
    usage: string;
    description: string;
    flags?: [string, string][];
    examples?: string[];
    valueFlags?: string[];
    run(ctx: CommandContext): void | Promise<void>;
}
